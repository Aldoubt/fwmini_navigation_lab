cmake_minimum_required(VERSION 3.8)
project(fwmini_chassis_adapter)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yhs_can_interfaces REQUIRED)
add_executable(${PROJECT_NAME}_node src/node.cpp)
target_compile_features(${PROJECT_NAME}_node PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME}_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME}_node PRIVATE include)
ament_target_dependencies(${PROJECT_NAME}_node rclcpp geometry_msgs std_srvs yhs_can_interfaces)
install(TARGETS ${PROJECT_NAME}_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core test/test_core.cpp)
  target_compile_features(test_core PRIVATE cxx_std_17)
  target_include_directories(test_core PRIVATE include)
  add_test(NAME core COMMAND test_core)
  find_package(launch_testing_ament_cmake REQUIRED)
  add_launch_test(test/test_ros.py TIMEOUT 60)
endif()
ament_package()
